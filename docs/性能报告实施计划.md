# 性能报告集成实施计划

## 一、概述

**目标:** 在测试报告中集成性能数据，支持系统性能和指定进程的性能监控

**前置条件:** 必须先实现串行/并行执行控制，只有串行模式才进行性能数据采集

**范围:**
- 任务执行模式: 串行 (concurrency=1) / 并行 (concurrency>1)
- 串行执行时，每个用例执行期间的性能数据
- 系统性能: CPU、内存、负载
- 进程性能: CPU、内存、文件描述符（按关键字选定）

class Agent:
    def __init__(self, config_path='config.yaml'):
        # ... existing code ...
        self.pending_tasks = []          # 待执行任务队列
        self.running_tasks = {}          # {result_id: case_info} 正在执行的任务
        self.max_concurrency = 10        # Agent 最大并发数

    def _handle_task_assign(self, data):
        """处理任务下发"""
        result_id = data.get('result_id')
        concurrency = data.get('concurrency', 1)  # 新增

        if concurrency == 1:
            # 串行模式：加入待执行队列
            self.pending_tasks.append(data)
            self._process_next_serial()
        else:
            # 并行模式：直接执行
            self._start_task(data)

    def _process_next_serial(self):
        """串行模式：处理下一个任务"""
        if not self.pending_tasks or len(self.running_tasks) >= 1:
            return

        task = self.pending_tasks.pop(0)
        self._start_task(task)

    def _start_task(self, task_data):
        """启动任务执行"""
        result_id = task_data.get('result_id')
        self.running_tasks[result_id] = task_data

        thread = threading.Thread(target=self._execute_task, args=(task_data, result_id))
        thread.daemon = True
        thread.start()

    def _execute_task(self, task_data, result_id):
        """执行单个任务"""
        # ... existing execution code ...

        # 任务完成回调
        self._on_task_complete(result_id, exec_result)

    def _on_task_complete(self, result_id, result):
        """任务完成处理"""
        # 从运行中移除
        if result_id in self.running_tasks:
            del self.running_tasks[result_id]

        # 上报结果
        self._report_result(task_data['task_id'], result_id, result)

        # 如果是串行模式，继续下一个任务
        if task_data.get('concurrency') == 1:
            self._process_next_serial()

                'concurrency': task_concurrency,
                # ... other fields ...
            }
            socketio.emit('task_assign', dispatch_data, namespace='/ws/agent')
            dispatched_cases.append(case_info['case_id'])

    # 更新任务状态
    task.status = 'running'
    task.started_at = datetime.utcnow()

### Step 5: Agent 端修改

**文件:** `agent/collector.py`

```python
import psutil
import os

class PerformanceCollector:
    def __init__(self):
        self.running = False
        self.thread = None
        self.send_callback = None
        self.task_id = None
        self.result_id = None  # 新增
        self.process_keyword = None  # 新增
        self.interval = 5
        self._perf_samples = []  # 新增：存储采样用于汇总

    def start(self, task_id, send_callback, result_id=None, process_keyword=None):
        self.running = True
        self.send_callback = send_callback
        self.task_id = task_id
        self.result_id = result_id
        self.process_keyword = process_keyword
        self._perf_samples = []
        self.thread = threading.Thread(target=self._collect_loop)
        self.thread.daemon = True
        self.thread.start()

    def get_summary(self):
        """计算性能汇总"""
        if not self._perf_samples:
            return None

        system_cpus = [s['system']['cpu'] for s in self._perf_samples if s.get('system')]
        system_memories = [s['system']['memory'] for s in self._perf_samples if s.get('system')]
        system_loads = [max(s['system']['load_avg']) for s in self._perf_samples if s.get('system')]

        summary = {
            'system': {
                'cpu_avg': sum(system_cpus) / len(system_cpus) if system_cpus else 0,
                'cpu_max': max(system_cpus) if system_cpus else 0,
                'memory_avg': sum(system_memories) / len(system_memories) if system_memories else 0,
                'memory_max': max(system_memories) if system_memories else 0,
                'load_avg_max': max(system_loads) if system_loads else 0
            }
        }

        # 进程汇总
        process_samples = [s.get('process') for s in self._perf_samples if s.get('process')]
        if process_samples:
            proc_cpus = [p['cpu'] for p in process_samples]
            proc_memories = [p['memory'] for p in process_samples]
            proc_fds = [p['fd_count'] for p in process_samples]
            summary['process'] = {
                'keyword': self.process_keyword,
                'cpu_avg': sum(proc_cpus) / len(proc_cpus),
                'cpu_max': max(proc_cpus),
                'memory_avg': sum(proc_memories) / len(proc_memories),
                'memory_max': max(proc_memories),
                'fd_count_max': max(proc_fds)
            }

        return summary

    def _find_target_process(self):
        """根据关键字查找目标进程"""
        if not self.process_keyword:
            return None

        candidates = []
        for proc in psutil.process_iter(['pid', 'name', 'cmdline', 'cpu_percent']):
            try:
                cmdline = ' '.join(proc.info.get('cmdline') or [])
                if self.process_keyword in cmdline:
                    candidates.append(proc)
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                continue

        if not candidates:
            return None

        # 取 CPU 使用率最高的
        return max(candidates, key=lambda p: p.cpu_percent())

    def _collect_loop(self):
        try:
            import psutil
        except ImportError:
            return

        while self.running:
            try:
                # 系统性能
                cpu = psutil.cpu_percent()
                memory = psutil.virtual_memory().percent
                try:
                    load_avg = os.getloadavg()
                except Exception:
                    load_avg = (0, 0, 0)

                # 进程性能
                process_info = None
                if self.process_keyword:
                    target = self._find_target_process()
                    if target:
                        try:
                            process_info = {
                                'keyword': self.process_keyword,
                                'pid': target.pid,
                                'cpu': target.cpu_percent(),
                                'memory': target.memory_percent(),
                                'fd_count': target.num_fds(),
                                'status': target.status()
                            }
                        except (psutil.NoSuchProcess, psutil.AccessDenied):
                            pass

                perf_data = {
                    'system': {
                        'cpu': cpu,
                        'memory': memory,
                        'load_avg': list(load_avg)
                    },
                    'process': process_info
                }

                self._perf_samples.append(perf_data)

                if self.send_callback and self.task_id:
                    self.send_callback(self.task_id, perf_data, self.result_id)

            except Exception as e:
                pass

            time.sleep(self.interval)
```

    # ... existing code ...

    # 获取性能配置
    process_keyword = task_data.get('process_keyword')

    def perf_callback(tid, data, rid=None):

                    'agent_id': self.agent_id,
                    'result_id': rid,
                    **data,
                    'timestamp': datetime.utcnow().isoformat()
